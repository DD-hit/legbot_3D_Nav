cmake_minimum_required(VERSION 3.16)
project(d1_odom_bringup)
find_package(ament_cmake REQUIRED)
find_package(legbot_runtime REQUIRED)

ament_target_dependencies(fastlio_odom_adapter legbot_runtime)
install(TARGETS fastlio_odom_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config urdf DESTINATION share/${PROJECT_NAME})
ament_package()
